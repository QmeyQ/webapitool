381: async def _keepalive_stream(
382:     request: Request,
383:     body: ChatRequest,
384:     ctx: RequestContext,
385:     heartbeat_interval: float = 15.0,
386: ):
387:     """网页 AI 首字节等待期（DeepSeek 深度思考可达 90s+）发 SSE 心跳注释行。
388: 
389:     OpenAI 兼容客户端忽略 ': ' 注释行；无心跳时 OpenClaw 等客户端会在
390:     空闲超时后断开并报"无法验证 AI"。生产者任务 + 队列隔离，
391:     心跳超时不会取消下游生成器。"""
392:     queue: asyncio.Queue = asyncio.Queue()
393:     _DONE = object()
394: 
395:     async def _produce():
396:         try:
397:             async for chunk in _usage_aware_stream(request, body, ctx):
398:                 await queue.put(chunk)
399:         except Exception as exc:
400:             await queue.put(exc)
401:         finally:
402:             await queue.put(_DONE)
403: 
404:     task = asyncio.create_task(_produce())
405:     try:
406:         while True:
407:             try:
408:                 item = await asyncio.wait_for(queue.get(),
409:                                               timeout=heartbeat_interval)
410:             except asyncio.TimeoutError:
411:                 yield ": keep-alive\n\n"
412:                 continue
413:             if item is _DONE:
414:                 break
415:             if isinstance(item, Exception):
416:                 raise item
417:             yield item
418:     finally:
419:         if not task.done():
420:             task.cancel()

423: async def _usage_aware_stream(
424:     request: Request,
425:     body: ChatRequest,
426:     ctx: RequestContext
427: ):
428:     """OpenClaw 等客户端依赖 stream_options.include_usage 终块（choices 为空、含 usage）。"""
429:     from browser_core import SSEFormatter

432:     prompt_chars = sum(len(str(m.get("content") or "")) for m in body.messages)
433:     completion_chars = 0
434:     async for chunk in _stream_with_lifecycle(request, body, ctx):
435:         if (include_usage and isinstance(chunk, str)

438:             usage = {

443:             yield finish_part + "\n\n" + SSEFormatter.pack_usage_chunk(usage) + "data: [DONE]\n\n"

452:                         if delta.get("content"):

459: async def _stream_with_lifecycle(
460:     request: Request,
461:     body: ChatRequest,
462:     ctx: RequestContext
463: ):
464:     """
465:     流式响应 + 完整生命周期管理
466:     
467:     关键修复：
468:     1. 生成器在独立线程执行（不阻塞事件循环）
469:     2. 主协程定期检查连接状态
470:     3. 通过队列传递数据
471:     """
472:     
473:     disconnect_task = None
474:     worker_thread = None
475:     acquired = False
476:     
477:     try:
478:         # ========== 0. 模型 → 站点路由 ==========

480:             try:
481:                 import browser_core as _bc
482:                 key = _bc.match_site_domain(
483:                     model, list(config_engine.sites.keys()))
484:                 if key:
485:                     global _last_used_site
486:                     with _last_site_lock:
487:                         _last_used_site = key
488:                     return key
489:                 # 未命中 model id（如 web-browser）：映射到最近一次
490:                 # 成功使用的站点。同一客户端（如 OpenClaw 标题生成）
491:                 # 的后续调用紧随主调用，跟最近站点最可靠；活动 tab
492:                 # 可能停在别的站点，会造成标题/正文内容错位。
493:                 with _last_site_lock:
494:                     if _last_used_site:
495:                         return _last_used_site
496:                 # 默认站点（browser_config.json DEFAULT_SITE，
497:                 # 未配置时取 sites.json 第一个键）
498:                 default_site = BrowserConstants.get("DEFAULT_SITE")
499:                 if default_site and default_site in config_engine.sites:
500:                     return default_site
501:                 try:
502:                     browser = get_browser(auto_connect=False)
503:                     tab = browser.get_active_tab()
504:                     if tab is not None:
505:                         return _bc.detect_tab_site_domain(
506:                             getattr(tab, "url", ""),
507:                             list(config_engine.sites.keys())) or ""
508:                 except Exception:
509:                     pass
510:                 return ""
511:             except Exception:
512:                 return ""

516:         # ========== 1. 获取锁（按站点分离，不同站点可并行） ==========
517:         # 等待上限 300s：同站点前一请求可能是 GLM/DeepSeek 深度思考（3-5 分钟），

521:                                                      site_key=site_key)

542:         browser.set_stop_checker(ctx.should_stop)

931:         if not isinstance(chunk, str):

948:                 break

951:                 choice = data["choices"][0]
952:                 delta = choice.get("delta", {})
953:                 content = delta.get("content", "")
954:                 if content:
955:                     collected_content.append(content)
956:                 for tc in delta.get("tool_calls") or []:
957:                     idx = tc.get("index", 0)

963:                     if fn.get("name"):

965:                     if fn.get("arguments"):
966:                         slot["arguments"] += fn["arguments"]
967:                 if choice.get("finish_reason"):
968:                     finish_reason = choice["finish_reason"]
969:         if error_data:
970:             break
971: 
972:     if error_data:
973:         return JSONResponse(content=error_data, status_code=500)
974: 
975:     full_content = "".join(collected_content)
976:     message = {"role": "assistant", "content": full_content}
977:     if collected_tool_calls:
978:         message["tool_calls"] = [
979:             {
980:                 "id": slot["id"] or f"call_{uuid.uuid4().hex[:24]}",
981:                 "type": "function",
982:                 "function": {"name": slot["name"],
983:                              "arguments": slot["arguments"] or "{}"},
984:             }
985:             for _, slot in sorted(collected_tool_calls.items())
